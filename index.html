// ============================================================ // Google Apps Script – Aufzeichnungen Einfüttern // ============================================================ const SHEET_NAME = "Einträge"; const HEADERS = [ "Mitarbeiter", "Datum", "Uhrzeit", "Startgewicht (kg)", "Endgewicht (kg)", "Grassilage (kg)", "Silphie/GPS (kg)", "Schrot (kg)", "Kartoffeln (kg)", "Mist (kg)", "Befüllung (kg)", "Mais (kg)", "Notizen" ]; function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, HEADERS.length) .setFontWeight("bold") .setBackground("#2a4d27") .setFontColor("#ffffff"); } return sheet; } function corsResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function formatDatum(val) { if (!val) return ""; if (val instanceof Date) { const d = String(val.getDate()).padStart(2, "0"); const m = String(val.getMonth() + 1).padStart(2, "0"); const y = val.getFullYear(); return d + "." + m + "." + y; } return String(val).trim(); } function formatUhrzeit(val) { if (!val) return ""; if (val instanceof Date) { const h = String(val.getHours()).padStart(2, "0"); const m = String(val.getMinutes()).padStart(2, "0"); return h + ":" + m; } const n = parseFloat(val); if (!isNaN(n) && n >= 0 && n < 1) { const totalMin = Math.round(n * 1440); const h = Math.floor(totalMin / 60); const mn = totalMin % 60; return String(h).padStart(2, "0") + ":" + String(mn).padStart(2, "0"); } const s = String(val).trim(); const match = s.match(/(\d{1,2}):(\d{2})/); if (match) return match[1].padStart(2, "0") + ":" + match[2]; return s; } function doGet(e) { try { const action = e.parameter.action || "get"; if (action === "get") { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const rows = []; for (let i = 1; i < data.length; i++) { const r = data[i]; if (!r[0]) continue; const uhrzeitRaw = sheet.getRange(i + 1, 3).getDisplayValue(); rows.push({ mitarbeiter: r[0], datum: formatDatum(r[1]), uhrzeit: uhrzeitRaw, startgewicht: r[3], endgewicht: r[4], grassilage: r[5], silphie_gps: r[6], schrot: r[7], kartoffeln: r[8], mist: r[9], befuellung: r[10], mais: r[11], notizen: r[12], }); } return corsResponse({ status: "ok", rows }); } if (action === "append") { const p = e.parameter; const sheet = getSheet(); sheet.appendRow([ p.mitarbeiter, p.datum, p.uhrzeit, p.startgewicht, p.endgewicht, p.grassilage, p.silphie_gps, p.schrot, p.kartoffeln, p.mist, p.befuellung, p.mais, p.notizen || "", ]); const lastRow = sheet.getLastRow(); const uc = sheet.getRange(lastRow, 3); uc.setValue(p.uhrzeit); uc.setNumberFormat("@"); return corsResponse({ status: "ok" }); } if (action === "clear") { const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (lastRow > 1) sheet.deleteRows(2, lastRow - 1); return corsResponse({ status: "ok" }); } return corsResponse({ status: "error", message: "Unbekannte Aktion" }); } catch(err) { return corsResponse({ status: "error", message: err.toString() }); } } function doPost(e) { return doGet(e); }
